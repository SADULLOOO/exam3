<script>

function updateHeartbeat(){

    fetch(
        "/heartbeat/",
        {
            method:"GET",
            credentials:"same-origin"
        }
    )

}


function updateLiveTime(){

    fetch("/get_live_time/")

    .then(
        response => response.json()
    )

    .then(data => {

        document
        .getElementById("timer")
        .innerText = data.seconds

    })

}


setInterval(
    updateHeartbeat,
    5000
)


setInterval(
    updateLiveTime,
    2000
)


updateHeartbeat()

updateLiveTime()
